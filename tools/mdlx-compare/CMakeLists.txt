cmake_minimum_required(VERSION 3.20)
project(wc3_mdlx_compare LANGUAGES CXX)
if(NOT WHITEOUT_SOURCE)
  message(FATAL_ERROR "Set WHITEOUT_SOURCE to the pinned WhiteoutLib checkout")
endif()
set(WHITEOUT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHITEOUT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHITEOUT_ENABLE_CASC OFF CACHE BOOL "" FORCE)
set(WHITEOUT_ENABLE_MPQ OFF CACHE BOOL "" FORCE)
set(WHITEOUT_ENABLE_DATABASE OFF CACHE BOOL "" FORCE)
set(WHITEOUT_USE_ZLIBNG OFF CACHE BOOL "" FORCE)
set(WHITEOUT_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(WHITEOUT_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHITEOUT_SOURCE}" whiteout)
add_executable(whiteout-adapter adapter.cpp)
target_compile_features(whiteout-adapter PRIVATE cxx_std_20)
target_link_libraries(whiteout-adapter PRIVATE whiteout_lib)
